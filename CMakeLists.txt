cmake_minimum_required(VERSION 3.21)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DMXPULSE_DEFAULT_VERSION LIMIT_COUNT 1)
set(DMXPULSE_VERSION "${DMXPULSE_DEFAULT_VERSION}" CACHE STRING "Numeric application version")
set(DMXPULSE_BUILD_NUMBER "1" CACHE STRING "Incrementing build number (the macOS bundle version)")
project(DMXPulse VERSION ${DMXPULSE_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
# Shared Pixel Stereo code (the look), pinned as a submodule
set(PXS_BUILD_TESTS OFF)
add_subdirectory(third_party/pxs-qt EXCLUDE_FROM_ALL)
include(CTest)
function(dmx_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
set(DMXPULSE_SOURCES src/window.cpp src/window.h)
function(dmx_application target entry bundle_id)
  qt_add_executable(${target} MACOSX_BUNDLE ${entry} ${DMXPULSE_SOURCES})
  target_include_directories(${target} PRIVATE src)
  target_link_libraries(${target} PRIVATE Qt6::Widgets Qt6::Network pxs::theme pxs::dmxnet pxs::dmxview)
  target_compile_definitions(${target} PRIVATE DMXPULSE_VERSION="${PROJECT_VERSION}")
  set_target_properties(${target} PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "${bundle_id}" MACOSX_BUNDLE_BUNDLE_VERSION "${DMXPULSE_BUILD_NUMBER}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" WIN32_EXECUTABLE TRUE)
  qt_add_resources(${target} "${target}_notices" PREFIX "/legal" FILES LICENSE THIRD-PARTY-NOTICES.md licenses/LGPL-3.0.txt licenses/GPL-3.0.txt)
  dmx_warnings(${target})
endfunction()
dmx_application(DMXPulse src/main.cpp fr.pixelstereo.dmxpulse)
install(TARGETS DMXPulse BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  install(FILES packaging/linux/fr.pixelstereo.dmxpulse.desktop DESTINATION share/applications)
  install(FILES docs/mark.svg DESTINATION share/icons/hicolor/scalable/apps RENAME fr.pixelstereo.dmxpulse.svg)
  install(FILES LICENSE THIRD-PARTY-NOTICES.md DESTINATION share/doc/dmxpulse)
  set(CPACK_PACKAGE_NAME dmxpulse)
  set(CPACK_PACKAGE_VENDOR PixelStereo)
  set(CPACK_PACKAGE_CONTACT "PixelStereo")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Art-Net and sACN network DMX monitor")
  set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_SECTION net)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins")
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  set(CPACK_GENERATOR DEB)
  include(CPack)
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  qt_add_executable(window_test tests/window_test.cpp ${DMXPULSE_SOURCES})
  target_include_directories(window_test PRIVATE src)
  target_link_libraries(window_test PRIVATE Qt6::Test Qt6::Widgets Qt6::Network pxs::theme pxs::dmxnet pxs::dmxview)
  dmx_warnings(window_test)
  add_test(NAME window COMMAND window_test -platform offscreen -o "${CMAKE_CURRENT_BINARY_DIR}/window-test.txt,txt")
  set_tests_properties(window PROPERTIES TIMEOUT 90)
endif()
