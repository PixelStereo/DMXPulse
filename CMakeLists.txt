cmake_minimum_required(VERSION 3.21)
project(DMXPulse VERSION 1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
include(CTest)
function(dmx_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
qt_add_executable(DMXPulse MACOSX_BUNDLE src/main.cpp src/window.cpp src/window.h src/protocol.cpp src/protocol.h)
target_link_libraries(DMXPulse PRIVATE Qt6::Widgets Qt6::Network)
set_target_properties(DMXPulse PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER fr.pixelstereo.dmxpulse WIN32_EXECUTABLE TRUE)
dmx_warnings(DMXPulse)
install(TARGETS DMXPulse BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  qt_add_executable(protocol_test tests/protocol_test.cpp src/protocol.cpp)
  target_include_directories(protocol_test PRIVATE src)
  target_link_libraries(protocol_test PRIVATE Qt6::Test Qt6::Network)
  dmx_warnings(protocol_test)
  add_test(NAME protocol COMMAND protocol_test)
  qt_add_executable(window_test tests/window_test.cpp src/window.cpp src/window.h src/protocol.cpp)
  target_include_directories(window_test PRIVATE src)
  target_link_libraries(window_test PRIVATE Qt6::Test Qt6::Widgets Qt6::Network)
  dmx_warnings(window_test)
  add_test(NAME window COMMAND window_test -platform offscreen)
endif()
