cmake_minimum_required(VERSION 3.21)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DMXPULSE_DEFAULT_VERSION LIMIT_COUNT 1)
set(DMXPULSE_VERSION "${DMXPULSE_DEFAULT_VERSION}" CACHE STRING "Numeric application version")
set(DMXPULSE_BUILD_NUMBER "1" CACHE STRING "Android/iOS incrementing build number")
project(DMXPulse VERSION ${DMXPULSE_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
include(CTest)
option(DMXPULSE_BUILD_MOBILE_TESTS "Build on-device smoke-test application" OFF)
function(dmx_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
set(DMXPULSE_SOURCES src/window.cpp src/window.h src/protocol.cpp src/protocol.h src/platform.cpp src/platform.h src/levelcontrol.cpp src/levelcontrol.h)
function(dmx_application target entry bundle_id)
  qt_add_executable(${target} MACOSX_BUNDLE ${entry} ${DMXPULSE_SOURCES})
  target_include_directories(${target} PRIVATE src)
  target_link_libraries(${target} PRIVATE Qt6::Widgets Qt6::Network)
  target_compile_definitions(${target} PRIVATE DMXPULSE_VERSION="${PROJECT_VERSION}")
  set_target_properties(${target} PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "${bundle_id}" MACOSX_BUNDLE_BUNDLE_VERSION "${DMXPULSE_BUILD_NUMBER}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" WIN32_EXECUTABLE TRUE)
  if(ANDROID)
    set(android_template "${CMAKE_CURRENT_BINARY_DIR}/${target}-android")
    file(MAKE_DIRECTORY "${android_template}")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/packaging/android/AndroidManifest.xml" manifest)
    string(REPLACE "package=\"fr.pixelstereo.dmxpulse\"" "package=\"${bundle_id}\"" manifest "${manifest}")
    file(WRITE "${android_template}/AndroidManifest.xml" "${manifest}")
    set_target_properties(${target} PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR "${android_template}" QT_ANDROID_PACKAGE_NAME "${bundle_id}" QT_ANDROID_VERSION_CODE "${DMXPULSE_BUILD_NUMBER}" QT_ANDROID_VERSION_NAME "${PROJECT_VERSION}" QT_ANDROID_MIN_SDK_VERSION 28 QT_ANDROID_TARGET_SDK_VERSION 35)
  elseif(IOS)
    set_target_properties(${target} PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/ios/Info.plist.in" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2" XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${bundle_id}")
    if(DMXPULSE_IOS_SIGNING)
      set_target_properties(${target} PROPERTIES XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/ios/DMXPulse.entitlements")
    endif()
  endif()
  qt_add_resources(${target} "${target}_notices" PREFIX "/legal" FILES LICENSE THIRD-PARTY-NOTICES.md licenses/LGPL-3.0.txt licenses/GPL-3.0.txt)
  dmx_warnings(${target})
endfunction()
